import * as React from "react";
import { Card } from "@/components/ui/card";
import { Coffee, DollarSign, Car, TrendingUp, TrendingDown } from "lucide-react";
import { cn } from "@/lib/utils";

interface WeatherCardProps {
  temperature: number;
  city: string;
  className?: string;
}

interface CurrencyCardProps {
  rate: number;
  change: number;
  className?: string;
}

interface TrafficCardProps {
  status: "light" | "moderate" | "heavy";
  estimatedTime: string;
  className?: string;
}

const WeatherCard = React.forwardRef<HTMLDivElement, WeatherCardProps>(
  ({ temperature = 24, city = "São Paulo", className }, ref) => (
    <Card
      ref={ref}
      variant="lifted"
      className={cn(
        "bg-gradient-to-br from-blue-50 to-blue-100 dark:from-blue-950 dark:to-blue-900 border-blue-200 dark:border-blue-800",
        className
      )}
    >
      <div className="p-6">
        <div className="flex items-center gap-3 mb-4">
          <Coffee className="h-6 w-6 text-blue-600 dark:text-blue-400" />
          <h3 className="text-lg font-semibold text-blue-900 dark:text-blue-100">
            Clima
          </h3>
        </div>
        <div className="space-y-2">
          <div className="text-3xl font-bold text-blue-900 dark:text-blue-100">
            {temperature}°C
          </div>
          <div className="text-sm text-blue-700 dark:text-blue-300">
            {city}
          </div>
        </div>
      </div>
    </Card>
  )
);
WeatherCard.displayName = "WeatherCard";

const CurrencyCard = React.forwardRef<HTMLDivElement, CurrencyCardProps>(
  ({ rate = 5.25, change = 0.15, className }, ref) => (
    <Card
      ref={ref}
      variant="lifted"
      className={cn(
        "bg-gradient-to-br from-green-50 to-green-100 dark:from-green-950 dark:to-green-900 border-green-200 dark:border-green-800",
        className
      )}
    >
      <div className="p-6">
        <div className="flex items-center gap-3 mb-4">
          <DollarSign className="h-6 w-6 text-green-600 dark:text-green-400" />
          <h3 className="text-lg font-semibold text-green-900 dark:text-green-100">
            USD/BRL
          </h3>
        </div>
        <div className="space-y-2">
          <div className="text-3xl font-bold text-green-900 dark:text-green-100">
            R$ {rate.toFixed(2)}
          </div>
          <div className="flex items-center gap-1">
            {change >= 0 ? (
              <TrendingUp className="h-4 w-4 text-green-600 dark:text-green-400" />
            ) : (
              <TrendingDown className="h-4 w-4 text-red-600 dark:text-red-400" />
            )}
            <span
              className={cn(
                "text-sm font-medium",
                change >= 0
                  ? "text-green-600 dark:text-green-400"
                  : "text-red-600 dark:text-red-400"
              )}
            >
              {change >= 0 ? "+" : ""}{change.toFixed(2)}%
            </span>
          </div>
        </div>
      </div>
    </Card>
  )
);
CurrencyCard.displayName = "CurrencyCard";

const TrafficCard = React.forwardRef<HTMLDivElement, TrafficCardProps>(
  ({ status = "moderate", estimatedTime = "25 min", className }, ref) => {
    const getStatusColor = (status: string) => {
      switch (status) {
        case "light":
          return "text-green-600 dark:text-green-400";
        case "moderate":
          return "text-yellow-600 dark:text-yellow-400";
        case "heavy":
          return "text-red-600 dark:text-red-400";
        default:
          return "text-gray-600 dark:text-gray-400";
      }
    };

    const getStatusText = (status: string) => {
      switch (status) {
        case "light":
          return "Trânsito Livre";
        case "moderate":
          return "Trânsito Moderado";
        case "heavy":
          return "Trânsito Intenso";
        default:
          return "Trânsito";
      }
    };

    const getBgColor = (status: string) => {
      switch (status) {
        case "light":
          return "from-green-50 to-green-100 dark:from-green-950 dark:to-green-900 border-green-200 dark:border-green-800";
        case "moderate":
          return "from-yellow-50 to-yellow-100 dark:from-yellow-950 dark:to-yellow-900 border-yellow-200 dark:border-yellow-800";
        case "heavy":
          return "from-red-50 to-red-100 dark:from-red-950 dark:to-red-900 border-red-200 dark:border-red-800";
        default:
          return "from-gray-50 to-gray-100 dark:from-gray-950 dark:to-gray-900 border-gray-200 dark:border-gray-800";
      }
    };

    return (
      <Card
        ref={ref}
        variant="lifted"
        className={cn(
          "bg-gradient-to-br",
          getBgColor(status),
          className
        )}
      >
        <div className="p-6">
          <div className="flex items-center gap-3 mb-4">
            <Car className={cn("h-6 w-6", getStatusColor(status))} />
            <h3 className={cn("text-lg font-semibold", getStatusColor(status))}>
              Trânsito
            </h3>
          </div>
          <div className="space-y-2">
            <div className={cn("text-lg font-semibold", getStatusColor(status))}>
              {getStatusText(status)}
            </div>
            <div className={cn("text-sm", getStatusColor(status))}>
              Tempo estimado: {estimatedTime}
            </div>
          </div>
        </div>
      </Card>
    );
  }
);
TrafficCard.displayName = "TrafficCard";

interface StackedCardsProps {
  className?: string;
}

const StackedCards = React.forwardRef<HTMLDivElement, StackedCardsProps>(
  ({ className }, ref) => (
    <div
      ref={ref}
      className={cn("relative w-full max-w-sm mx-auto", className)}
    >
      <div className="relative space-y-4">
        {/* Third card (bottom) - Traffic */}
        <div className="transform translate-y-8 translate-x-4 rotate-2 opacity-70">
          <TrafficCard status="moderate" estimatedTime="25 min" />
        </div>
        
        {/* Second card (middle) - Currency */}
        <div className="transform translate-y-4 translate-x-2 rotate-1 opacity-85 relative z-10">
          <CurrencyCard rate={5.25} change={0.15} />
        </div>
        
        {/* First card (top) - Weather */}
        <div className="transform relative z-20">
          <WeatherCard temperature={24} city="São Paulo" />
        </div>
      </div>
    </div>
  )
);
StackedCards.displayName = "StackedCards";

function Component() {
  return (
    <div className="min-h-screen bg-background p-8 flex items-center justify-center">
      <StackedCards />
    </div>
  );
}

export default Component;
